/** 取一个正整数；非法取值返回 undefined。 */
function readPositiveInt(raw: unknown): number | undefined {
  const value = Number(raw);
  return Number.isFinite(value) && value > 0 ? value : undefined;
}

/** 归一化降级档；缺失或非法取值退回默认的 notify-only。 */
function normalizeDegradation(raw: unknown): Degradation {
  return raw === "strict" || raw === "read-only-allow" || raw === "notify-only" ? raw : "notify-only";
}

const DEGRADATION: Degradation = normalizeDegradation(GATE_CONFIG.degradation);

/**
 * 闸门问什么：写档与执行档要不要阻塞等人点按。
 *
 * `all`（缺省，闸门的原始形态）——写档与执行档都问；
 * `critical-only`——只问危险命令档，其余写/执行调用照跑（仍然上报，行里看得到工具在跑）。
 *
 * 「只问危险命令」不是「不问」：`critical` 在任何档位下都要问，且应用不可达时一律拒绝。
 * 缺省取 `all`：老应用不会注入这个字段，缺省必须与它期待的行为一致。
 */
type AskScope = "all" | "critical-only";

/** 归一化适用范围；缺失或非法取值退回 `all`。 */
function normalizeAskScope(raw: unknown): AskScope {
  return raw === "critical-only" ? "critical-only" : "all";
}

const ASK_SCOPE: AskScope = normalizeAskScope(GATE_CONFIG.askScope);

/** 该档位要不要问刘海。 */
function asksByScope(tier: ApprovalTier): boolean {
  return tier === "critical" || ASK_SCOPE === "all";
}

/** 从未知值里取布尔字段。 */
function readBool(source: unknown, key: string): boolean {
  if (typeof source !== "object" || source === null || !(key in source)) {
    return false;
  }
  return (source as Record<string, unknown>)[key] === true;
}

/** 从未知值里取字符串字段，取不到时返回 undefined。 */
function readString(source: unknown, key: string): string | undefined {
  if (typeof source !== "object" || source === null || !(key in source)) {
    return undefined;
  }
  const value = (source as Record<string, unknown>)[key];
  return typeof value === "string" && value.length > 0 ? value : undefined;
}

/** 从未知值里取对象字段。 */
function readObject(source: unknown, key: string): Record<string, unknown> | undefined {
  if (typeof source !== "object" || source === null || !(key in source)) {
    return undefined;
  }
  const value = (source as Record<string, unknown>)[key];
  return typeof value === "object" && value !== null ? (value as Record<string, unknown>) : undefined;
}

// 注意：omp 的扩展模块在**整个进程内只求值一次**，下面这些模块级变量是所有会话
// （根会话与各个 subAgent 会话）共用的。它们只记录「根会话」的身份，子代理上下文
// 只读不写——否则子代理的 session_start 会把根会话身份冲掉，后续事件（含根会话自己的）
// 都会带着子代理的会话 id 上报（真实踩过：一次嵌套派发里孙代理的 id 顶掉了根会话 id）。
let rootSessionFile: string | undefined;
let rootSessionId: string | undefined;
let pid: number | undefined;
let tty: string | undefined;

// 串行发送，保证事件顺序；应用不在时也不会堆积。
let chain: Promise<boolean> = Promise.resolve(true);

/** 取当前进程所在 tty，用于从 notch 聚焦终端。 */
function detectTty(): string | undefined {
  try {
    const out = execSync(`ps -p ${process.pid} -o tty=`, { encoding: "utf8", timeout: 800 }).trim();
    // 无 tty 时 ps 会给出 `?` / `??` / `-`，这些都不是可聚焦的终端名
    if (/^\/?[a-z]+[0-9]+$/.test(out) || /^\/?ttys[0-9]+$/.test(out)) {
      return out.startsWith("/dev/") ? out.slice("/dev/".length) : out;
    }
  } catch {
    // 忽略：拿不到 tty 只是无法从 notch 聚焦终端。
  }
  return undefined;
}

/**
 * 会话文件是否是「根会话」形状：`<时间戳>_<uuid>.jsonl`。
 *
 * 子代理的会话文件形如 `<实例名>.jsonl`（没有时间戳前缀），与根会话一眼可分——比
 * `ctx.sessionManager.getHeader()` 可靠：会话头要等会话落盘才存在，子代理的
 * `session_start` 期间常常读不到，只看会话头会把子代理误判成根会话（也是真实踩过的坑）。
 */
function looksLikeRootSessionFile(path: string | undefined): boolean {
  if (!path) {
    return false;
  }
  const last = path.split("/").pop();
  const base = last && last.endsWith(".jsonl") ? last.slice(0, -".jsonl".length) : last;
  return base !== undefined && /^[0-9T:.-]+Z_[0-9a-f-]{8,}$/.test(base);
}

/** 读上下文里的会话文件路径（读取失败返回 undefined）。 */
function currentSessionFilePath(ctx: ExtensionContextLike | undefined): string | undefined {
  try {
    const file = ctx?.sessionManager?.getSessionFile?.();
    return typeof file === "string" && file.startsWith("/") ? file : undefined;
  } catch {
    return undefined;
  }
}

/**
 * 记下根会话身份。只有根会话的上下文才允许写入。
 *
 * 模块级变量在进程内共享，子代理上下文只能读、不能写：一旦被覆盖，
 * 后续所有事件都会带着错误的会话 id 上报。
 */
function bindRootSession(ctx: ExtensionContextLike | undefined): void {
  const file = currentSessionFilePath(ctx);
  if (file) {
    rootSessionFile = file;
  }
  try {
    const id = ctx?.sessionManager?.getSessionId?.();
    if (typeof id === "string" && id.length > 0) {
      rootSessionId = id;
    }
  } catch {
    // 保持既有值
  }
}

/**
 * 会话头是否带 parentSession。
 *
 * omp 用它标注「这个会话由某个父会话派生」（子 Agent、持久化复活的子代理），
 * 是区分「子 Agent」与「headless 根会话」的唯一权威字段。
 */
function hasParentSession(ctx: ExtensionContextLike | undefined): boolean {
  try {
    const header = ctx?.sessionManager?.getHeader?.();
    const parent = readString(header, "parentSession");
    return parent !== undefined;
  } catch {
    // 取不到会话头时按「不是子会话」处理，宁可多报一个会话，也不静默丢失。
    return false;
  }
}

/**
 * 是否是应用要展示的「根会话」（用户眼前的那一个）。
 *
 * 不能只看 `hasUI`：omp 的子 Agent 与 headless 根会话（`omp -p`、`--mode json|rpc`）
 * 都是 `hasUI=false` + `mode="print"`，只按 hasUI 过滤会把 headless 会话一并丢掉。
 * 子 Agent 的权威判据是会话头里的 `parentSession`；TUI 一定是根会话（子 Agent 不
 * 带 UI），因此 hasUI 为真时直接放行，避免 `--fork` 出来的根会话被误判成子会话。
 */
function isRootSession(ctx: ExtensionContextLike | undefined): boolean {
  if (ctx?.hasUI === true || ctx?.mode === "tui") {
    return true;
  }
  if (hasParentSession(ctx)) {
    return false;
  }
  // 会话头读不到（子代理 session_start 期间常见）时，退回到会话文件命名形状。
  const file = currentSessionFilePath(ctx);
  return file === undefined || looksLikeRootSessionFile(file);
}

// ---------------------------------------------------------------------------
// 审批闸门
//
// 把 tool_call 变成「阻塞闸门」：需要问的档位先把请求投给刘海应用，等它回
// {"decision":"allow"|"deny"}，再决定放行还是 {block:true, reason}。
// 三条硬约束：
//   ① 永不抛错——抛错会被 omp 记成 `Extension ... failed` 并阻塞工具；
//   ② 审批**不进 chain**——一个待批堵住整条上报链，比丢一条状态严重得多；
//   ③ 写完后不 end()，收到完整 JSON 再 destroy()——实测（E1/E2/E3）写完就收尾会丢应答。
// ---------------------------------------------------------------------------

/** 审批档位：allow（只读/协议，直接放行）/ write（落盘）/ exec（执行）/ critical（危险命令）。 */
type ApprovalTier = "allow" | "write" | "exec" | "critical";

/**
 * never-ask 白名单：只读与协议类工具，连 socket 都不发。
 *
 * 取向是「默认询问」——名单之外的一律问，因此 omp 将来新增的工具会自动落入「要问」，
 * 漏的是「多问」而不是「漏问」。`yield` 必须留在名单里：拒 yield 会让子代理无法交付。
 */
const NEVER_ASK_TOOLS = new Set([
  "read",
  "glob",
  "grep",
  "ask",
  "todo",
  "yield",
  "think",
  "ast-grep",
  "checkpoint",
]);

/** 前缀形态的 never-ask（记忆类工具名带动作后缀）。 */
const NEVER_ASK_PREFIXES = ["memory-"];

/** write 档：会落盘但不执行命令的工具（用于 approval_kind 展示与降级档判定）。 */
const WRITE_TIER_TOOLS = new Set(["write", "edit", "ast-edit"]);

/**
 * 危险命令模式：内联一份精简名单，等价 omp 的 CRITICAL_BASH_PATTERNS
 * （权威源 `oh-my-pi/packages/coding-agent/src/tools/bash.ts`，omp 18.2.6 共 21 条）。
 *
 * 不 import omp 内部模块：那会随包结构与导出变化而崩；这份名单随 omp 升级人工对齐。
 * 命中即标红，并且在应用不可达时**任何降级档都直接拒绝**。
 */
const CRITICAL_COMMAND_PATTERNS: readonly RegExp[] = [
  // 递归删除（选项可在 -r/-f 两侧，但必须带绝对根路径）。
  /\brm\s+(?:-\S+\s+)*(?:-[a-z]*[rRfF][a-z]*|--recursive|--force)\s+(?:-\S+\s+)*\//i,
  /\brm\s+(?:-\S+\s+)*--no-preserve-root\b/i,
  /\bsudo\s+rm\b/i,
  /\bchmod\s+-R\s+[0-7]+\s+\//i,
  /\bchmod\s+-R\s+[ugoa+\-=rwxXst,]+\s+\//,
  /\bchown\s+-R\s+\S+\s+\//i,
  // fork 炸弹。
  /:\(\)\s*\{\s*:\s*\|\s*:/i,
  // 磁盘 / 文件系统破坏。
  />\s*\/dev\/sd[a-z]/i,
  /\bmkfs(\.|\b)/i,
  /\bdd\s+if=.+of=\/dev\//i,
  /\bshred\s+\/dev\//i,
  /\bcryptsetup\b/i,
  // 系统配置破坏。
  />\s*\/etc\/(?:passwd|shadow|sudoers)\b/i,
  /\btee\s+(?:-a\s+)?\/etc\/(?:passwd|shadow|sudoers)\b/i,
  // 远程拉取后直接执行。
  /\b(?:curl|wget|fetch)\b[^|]*\|\s*(?:bash|sh|zsh|fish)\b/i,
  /(?:^|[\s;&|(])(?:bash|sh|zsh|source|\.)\s+<\(\s*(?:curl|wget|fetch)\b/i,
  /\beval\s+["'`]?\$\(\s*(?:curl|wget|fetch)\b|\beval\s+`\s*(?:curl|wget|fetch)\b/i,
  // 进程 / 主机控制。
  /\bkill\s+-9\s+1\b/,
  /(?:^|[\s;&|(])(?:shutdown|poweroff|reboot|halt)(?:\s|$|[;|&])/i,
  /(?:^|[\s;&|(])init\s+0\b/i,
  // 反弹 shell。
  /\bnc\b[^|;]*\s-[a-zA-Z]*[ec][a-zA-Z]*\s/i,
];

/** 闸门自检结果：探测失败即降级为只上报（绝不静默放行）。 */
let gateEnabled = true;
let gateDisabledReason: string | undefined;
let gateDisabledReported = false;
/** 诊断用：omp 生效的审批档位（`pi.pi.*` 不是扩展 API 契约，只作诊断，不作安全判定）。 */
let approvalMode: string | undefined;
/** 降级提示在每个进程里只弹一次（状态行每次刷新）。 */
let degradationNotified = false;

/** 该工具是否属于 never-ask（只读/协议）。 */
function isNeverAskTool(toolName: string): boolean {
  return NEVER_ASK_TOOLS.has(toolName)
    || NEVER_ASK_PREFIXES.some((prefix) => toolName.startsWith(prefix));
}

/** 从工具入参里取出可能含危险命令的文本（bash 的命令、eval 的代码）。 */
function commandTextOf(toolName: string, input: unknown): string | undefined {
  if (toolName === "bash") {
    return readString(input, "command");
  }
  if (toolName === "eval") {
    return readString(input, "code");
  }
  return undefined;
}

/** 该命令文本是否命中危险模式。 */
function isCriticalCommand(text: string | undefined): boolean {
  return text !== undefined && CRITICAL_COMMAND_PATTERNS.some((pattern) => pattern.test(text));
}

/** 给一次工具调用定档。 */
function classifyToolCall(toolName: string, input: unknown): ApprovalTier {
  if (isNeverAskTool(toolName)) {
    return "allow";
  }
  if (isCriticalCommand(commandTextOf(toolName, input))) {
    return "critical";
  }
  return WRITE_TIER_TOOLS.has(toolName) ? "write" : "exec";
}

  try {

  } catch {

  }

/**
 * 闸门是否拦这一条（只拦根会话）。
 *
 * 判据顺序：① 会话头带 `parentSession` → 子代理（omp 的权威字段）；② 会话文件名形状，
 * 复用本文件的 `looksLikeRootSessionFile`（根会话 `<时间戳>_<uuid>.jsonl`，子代理
 * `<实例名>.jsonl`）；③ 两者都拿不到时退回 `isRootSession`。都拿不到按根会话处理——
 * 宁可多问一次，也不漏掉用户眼前的会话。
 */
function isGateRootSession(ctx: ExtensionContextLike | undefined): boolean {
  if (hasParentSession(ctx)) {
    return false;
  }
  const file = currentSessionFilePath(ctx);
  if (file) {
    return looksLikeRootSessionFile(file);
  }
  return isRootSession(ctx);
}

/** 在 TUI 里留一行可见的降级提示（拿不到 UI 时静默——绝不因此抛错）。 */

/** 「降级放行」的审计事件：先排队，等下一次成功上报时补投（应用恢复后刘海能看到发生过什么）。 */
const degradedReports: Record<string, unknown>[] = [];
const MAX_DEGRADED_REPORTS = 20;

/** 记一条降级审计（超上限丢最旧的，避免内存无限增长）。 */
function recordDegradedReport(body: Record<string, unknown>): void {
  if (degradedReports.length >= MAX_DEGRADED_REPORTS) {
    degradedReports.shift();
  }
  degradedReports.push(body);
}

/** 把排队的降级事件补投给应用；投不出去就留回队列，下次再试。 */
function flushDegradedReports(): void {
  if (degradedReports.length === 0) {
    return;
  }
  const pending = degradedReports.splice(0, degradedReports.length);
  for (const body of pending) {
    void deliver(body).then((delivered) => {
      if (!delivered) {
        recordDegradedReport(body);
      }
    });
  }
}

/** 一次审批询问的结果。 */
/** 一次询问的结果。 */
interface ApprovalOutcome {
  kind: "decision" | "timeout" | "unreachable" | "aborted";
  decision?: string;
  reason?: string;
  /** `ask` 作答（`decision === "answer"`）；其余决定不带这个键。 */
  answers?: AskAnswers;
}

/**
 * 可取消的阻塞询问：`outcome` 是结局，`cancel()` 主动收掉 socket。
 * 影子 `ask` 在「原生先答」时用它把刘海那条撤干净（连接一关，应用就把卡片当
 * 外部已裁决：应用侧 ClaudeSessionMonitor 在 PostToolUse 上清待批）。
 */
interface DecisionRequest {
  outcome: Promise<ApprovalOutcome>;
  cancel: () => void;
}

/**
 * 归一化下行 `answers`：只保留「问题 id → 字符串数组」，其余一律丢掉。
 *
 * **键存在 = 该题已作答，值可以是空数组**：多选允许零勾选前移（omp 原生
 * `ask-dialog.ts` 的 Submit 页 Enter 直接提交，可不勾任何项），应用侧因此会回
 * `{"<id>": []}`。空数组必须原样保留——丢掉它会让「零选」被当成「未作答」，
 * 进而落进取消分支（见 `raceIslandAndNativeAsk` 的判定）。
 * 非字符串与空白项仍然过滤（它们不是合法的 label）。
 */
function parseAskAnswers(raw: unknown): AskAnswers | undefined {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) {
    return undefined;
  }
  const answers: AskAnswers = {};
  for (const [key, value] of Object.entries(raw as Record<string, unknown>)) {
    if (key.length === 0 || !Array.isArray(value)) {
      continue;
    }
    const labels = value.filter((entry): entry is string => typeof entry === "string" && entry.length > 0);
    answers[key] = labels;
  }
  return Object.keys(answers).length > 0 ? answers : undefined;
}

/**
 * 阻塞询问刘海：写一个 JSON，然后等应答。
 *
 * 与 send() 的差别（§5.2/§5.6）：独立 Promise、不复用 chain、写完后不 end()，
 * 累积 data 直到能 JSON.parse，决策 / 超时 / 断开后 destroy() 释放 fd。
 * 返回可取消形态：闸门只用 `requestDecision`（不需要中途撤下），影子 `ask` 用它
 * 在「原生先答」时把这条关掉。
 */
function startDecisionRequest(body: Record<string, unknown>, timeoutMs: number): DecisionRequest {
  const { promise, resolve } = Promise.withResolvers<ApprovalOutcome>();
  let buffer = "";
  let connected = false;
  let settled = false;
  let timer: ReturnType<typeof setTimeout> | undefined;
  let socket: NodeSocket | undefined;
  const done = (outcome: ApprovalOutcome) => {
    if (settled) {
      return;
    }
    settled = true;
    if (timer !== undefined) {
      clearTimeout(timer);
    }
    try {
      socket?.destroy();
    } catch {
      // 忽略：连接可能已经断开。
    }
    resolve(outcome);
  };
  try {
    socket = net.createConnection(SOCKET_PATH, () => {
      connected = true;
      try {
        socket?.write(JSON.stringify(body));
      } catch {
        done({ kind: "aborted" });
      }
    });
    timer = setTimeout(() => done({ kind: "timeout" }), timeoutMs);
    socket.on("data", (chunk: Buffer | string) => {
      buffer += chunk.toString("utf8");
      const text = buffer.trim();
      if (text.length === 0) {
        return;
      }
      try {
        const parsed = JSON.parse(text) as { decision?: unknown; reason?: unknown; answers?: unknown };
        const answers = parseAskAnswers(parsed.answers);
        done({
          kind: "decision",
          decision: typeof parsed.decision === "string" ? parsed.decision : undefined,
          reason: typeof parsed.reason === "string" ? parsed.reason : undefined,
          ...(answers === undefined ? {} : { answers }),
        });
      } catch {
        // 数据还没收全：继续累积，直到超时。
      }
    });
    socket.on("error", () => done(connected ? { kind: "aborted" } : { kind: "unreachable" }));
    socket.on("close", () => done(connected ? { kind: "aborted" } : { kind: "unreachable" }));
  } catch {
    done({ kind: "unreachable" });
  }
  return { outcome: promise, cancel: () => done({ kind: "aborted" }) };
}

/** 阻塞询问刘海（闸门用：不需要中途撤下）。 */
function requestDecision(body: Record<string, unknown>, timeoutMs: number): Promise<ApprovalOutcome> {
  return startDecisionRequest(body, timeoutMs).outcome;
}

/**
 * 按失败语义矩阵把询问结果折算成最终裁决。
 *
 * 「问了没答」= 拒绝（保安全，omp 在 yolo 下没有任何原生兜底）；
 * 「无人可问」= 按用户选的降级档（保可用性，且降级必须在 TUI 可见）。
 */
function resolveOutcome(
  tier: ApprovalTier,
  outcome: ApprovalOutcome,
): { allowed: boolean; reason?: string } {
  if (outcome.kind === "decision") {
    if (outcome.decision === "allow") {
      return { allowed: true };
    }
    return { allowed: false, reason: outcome.reason ?? "Denied on AgentIsland" };
  }
  if (outcome.kind === "timeout") {
    const seconds = Math.round(APPROVAL_TIMEOUT_MS / 1000);
    return { allowed: false, reason: `Approval request timed out — no answer on AgentIsland within ${seconds}s.` };
  }
  if (outcome.kind === "aborted") {
    // 已经问过人又被中断：不静默放行。
    return { allowed: false, reason: "AgentIsland is no longer available" };
  }
  // 应用从未运行：危险命令一律拒绝，其余看用户选的降级档。
  if (tier === "critical") {
    return {
      allowed: false,
      reason: "Blocked on AgentIsland: known-dangerous command while the approval gate is unreachable",
    };
  }
  switch (DEGRADATION) {
    case "strict":
      return { allowed: false, reason: "Blocked on AgentIsland: approval gate offline (strict)" };
    case "read-only-allow":
      // 只读档（never-ask）在闸门之前就放行了，走得到这里的都是 write/exec 档 → 拒绝。
      return { allowed: false, reason: "Blocked on AgentIsland: approval gate offline (read-only-allow)" };
    case "notify-only":
    default:
      return { allowed: true };
  }
}


/** 组装一条事件的完整信封；没有会话标识时返回 undefined（这一条不发）。 */
function buildBody(
  payload: IslandEvent,
  override?: { sessionId?: string; sessionFile?: string },
): Record<string, unknown> | undefined {
  // 模块级状态只保存**根会话**身份（子代理上下文只读不写），因此这里的缺省值就是根会话；
  // 子代理的请求由调用方通过 override 传它自己的会话身份。
  const sessionId = override?.sessionId ?? rootSessionId;
  const sessionFile = override?.sessionFile ?? rootSessionFile;
  if (!sessionId && !sessionFile) {
    return undefined;
  }
  return {
    session_id: sessionId ?? "unknown",
    cwd: typeof process.cwd === "function" ? process.cwd() : undefined,
    pid,
    tty,
    agent: AGENT,
    session_file: sessionFile,
    ...payload,
  };
}

/** 一次「连接 → 写入」；返回是否写入成功（用于决定要不要补投降级审计事件）。 */
function deliver(body: Record<string, unknown>): Promise<boolean> {
  const { promise, resolve } = Promise.withResolvers<boolean>();
  let wrote = false;
  let settled = false;
  let socket: NodeSocket | undefined;
  const done = () => {
    if (settled) {
      return;
    }
    settled = true;
    try {
      socket?.destroy();
    } catch {
      // 忽略：连接可能已经断开。
    }
    resolve(wrote);
  };
  try {
    socket = net.createConnection(SOCKET_PATH, () => {
      try {
        socket?.write(JSON.stringify(body));
        wrote = true;
      } catch {
        // 写入失败：按未送达处理。
      }
      socket?.end();
    });
    socket.setTimeout(400, done);
    socket.on("data", done);
    socket.on("error", done);
    socket.on("end", done);
    socket.on("close", done);
  } catch {
    done();
  }
  return promise;
}

/** 发送一条事件；永不抛错，永不阻塞调用方。 */
function send(
  payload: IslandEvent,
  override?: { sessionId?: string; sessionFile?: string },
): Promise<void> {
  const body = buildBody(payload, override);
  if (!body) {
    return Promise.resolve();
  }
  chain = chain
    .then(() => deliver(body))
    .then((delivered) => {
      // 上报成功后把降级期间放行的审计事件补投给应用。
      if (delivered) {
        flushDegradedReports();
      }
      return delivered;
    })
    .catch(() => false);
  return chain.then(() => undefined);
}

/**
 * 订阅本实例所在会话的共享总线上的子 Agent 频道。
 *
 * 扩展模块在进程内只求值一次，订阅到的总线覆盖整棵子代理树（含「子代理再派子代理」
 * 的孙代理）：所有总线事件都上报到根会话名下，由应用侧按 `parent_tool_call_id` 与
 * 子代理会话文件路径挂到对应的 task 卡片；子代理不再按自己的会话上报，避免污染会话身份。
 * 进度事件按「状态 + 当前工具」去重，避免流式进度把 socket 打满。
 */
function subscribeSubagentBus(pi: ExtensionApi): void {
  const bus = pi.events;
  if (!bus || typeof bus.on !== "function") {
    // 旧版 pi 没有共享总线：子 Agent 不上报，其余事件不受影响。
    return;
  }

  const lastProgressSignature = new Map<string, string>();

  /** 总线事件一律上报到根会话名下：子代理本身不是应用认识的会话。 */
  function rootOwner(): string | undefined {
    return rootSessionId;
  }

  bus.on(SUBAGENT_LIFECYCLE_CHANNEL, (data) => {
    const id = readString(data, "id");
    const owner = rootOwner();
    if (!id || !owner) {
      return;
    }
    void send(
      {
        event: "SubagentLifecycle",
        status: "processing",
        subagent_id: id,
        subagent_agent: readString(data, "agent") ?? id,
        subagent_status: readString(data, "status"),
        subagent_task: readString(data, "description"),
        parent_tool_call_id: readString(data, "parentToolCallId"),
        subagent_session_file: readString(data, "sessionFile"),
      },
      { sessionId: owner },
    );
  });

  bus.on(SUBAGENT_PROGRESS_CHANNEL, (data) => {
    const progress = readObject(data, "progress");
    const id = readString(progress, "id") ?? readString(data, "id");
    const owner = rootOwner();
    if (!id || !owner) {
      return;
    }
    const status = readString(progress, "status");
    const currentTool = readString(progress, "currentTool");
    const signature = `${status ?? ""}|${currentTool ?? ""}`;
    if (lastProgressSignature.get(id) === signature) {
      return;
    }
    lastProgressSignature.set(id, signature);
    void send(
      {
        event: "SubagentProgress",
        status: "processing",
        subagent_id: id,
        subagent_agent: readString(progress, "agent") ?? readString(data, "agent") ?? id,
        subagent_status: status,
        subagent_current_tool: currentTool,
        subagent_task: readString(progress, "task") ?? readString(data, "task"),
        parent_tool_call_id: readString(data, "parentToolCallId"),
        subagent_session_file: readString(data, "sessionFile"),
      },
      { sessionId: owner },
    );
  });
}

// 影子 `ask`：让 omp 的交互式提问能在刘海上「选择并回复」
//
// 为什么必须是同名影子工具：原生 `ask` 的提问界面属于 omp 自己的 UI 上下文
// （`AskTool.execute` 走 `context.ui.askDialog`，ask.ts:824-1165），扩展没有「从外部
// 作答」的入口；`tool_approval_requested/resolved` 则只是 observability（wrapper.ts
// 只 emit、丢弃返回值）。而 omp 的工具注册表是**后写覆盖**：扩展注册同名 `ask` 会顶掉
// 内置实现（sdk.ts:2945-2950），内置实现仍留在 nativeToolsByName 里供 `ctx.invokeTool`
// 委托（sdk.ts:2952-2956；文档 docs/extensions.md「Delegating to a native built-in」）。
//
// execute 并发两条路，先答者胜：
//   ① 刘海：阻塞询问（复用闸门的 socket 写法），上行带完整 questions 与
//      expects_response:true；
//   ② 原生：`ctx.invokeTool(params, { signal })` —— 保留终端作答能力。
// ①先答 → abort ②的 signal：原生 `untilAborted` 立刻抛 AbortError，被 ask.ts:1034-1036
// 折成 `ToolAbortError("Ask input was cancelled")`（**不会** context.abort()，本轮不终止），
// 同时 `#presentDialog` 的 abort 监听把对话框 dispose、编辑器还回来
// （extension-ui-controller.ts:1272-1330）。②先答 → 关掉 ①的 socket。
//
// 不退化：拿不到会话身份、非 TUI 根会话、原生不存在、应用不可达/超时 → 一律原生独占。
// 纪律：与闸门一致，永不抛错；唯一例外是「用户取消」——那是**按原生语义**主动抛
// `ToolAbortError`（ask.ts:1016 / 1125：原生取消 = `context.abort()` 后抛 `ToolAbortError`）。
// ---------------------------------------------------------------------------

/** 影子工具是否已注册（只有 omp 系才有原生 `ask` 可以委托，pi 0.85.1 没有 `ask`）。 */
let shadowAskRegistered = false;

/**
 * ask 作答预算。
 *
 * 读题、权衡、点选比「许可/拒绝」慢得多，而原生询问本身没有超时
 * （`ask.timeout` 默认 0 = 关闭，settings-schema.ts:2281-2297），所以它比闸门给得宽。
 * 但它必须**严格小于**服务端预算，四层从外到内是：
 *
 *   应用 pending TTL 330s  >  omp 服务端 toolCallTimeoutMs 300s  >  ask 客户端 240s  >  闸门客户端 120s
 *
 * 服务端那 300s 由安装器写进用户配置（`extensionHandlers.toolCallTimeoutMs`，
 * `OmpConfigInstaller.gateHandlerTimeoutMs`；消费方是扩展 `tool_call` handler 的
 * active-work 预算，settings-schema.ts:6094-6104）。
 * 取 240s 而不是 300s：**相等时「谁先到点」不可判定**——客户端以为自己仍是裁决者，
 * 服务端却可能同时 fail-closed，用户看到的是不可读的 `Extension … timed out`；
 * 而且相等这种配置在真机上要等满那个时长才能证伪。留 60s 余量后，「本扩展总是先说话、
 * 理由可读」不再依赖竞态。应用侧 pending TTL 仍取 330s（> 300s），TTL 不变。
 *
 * 可用 AGENT_ISLAND_ASK_TIMEOUT_MS 覆盖（验收靠它把超时改小）。
 */
const ASK_TIMEOUT_MS = readPositiveInt(process.env.AGENT_ISLAND_ASK_TIMEOUT_MS) ?? 240000;

 * **空数组是合法值**——多选的「零选」：`{"<id>": []}` 表示该题已作答且一项都没选，

/** 从入参里取数组字段（`readObject` 会把数组也当对象返回，故另开一个）。 */
function readArray(source: unknown, key: string): unknown[] | undefined {
  if (typeof source !== "object" || source === null || !(key in source)) {
    return undefined;
  }
  const value = (source as Record<string, unknown>)[key];
  return Array.isArray(value) ? value : undefined;
}

/**
 * 把工具入参归一成上行形状；任何一题缺 `id`/`question` 就返回 undefined（退回原生）。
 *
 * 字段语义逐条对齐原生 schema（ask.ts:51-71）：`id`/`question` 必有、`header` 可选、
 * `options[].label` 必有而 `description` 可选、`multi` 缺省按单选。
 * `free_text` **恒为 true**：原生每个问题都由 UI 自动补一行「Other (type your own)」
 * （ask.ts:42 的 OTHER_OPTION 与 prompts/tools/ask.md 的 critical 段），所以「能不能
 * 自由文本」在原生里恒真；上报成 false 会凭空砍掉一个原生有的入口。
 */
function normalizeAskQuestions(input: unknown): AskWireQuestion[] | undefined {
  const rawQuestions = readArray(input, "questions");
  if (rawQuestions === undefined || rawQuestions.length === 0) {
    return undefined;
  }
  const questions: AskWireQuestion[] = [];
  for (const rawQuestion of rawQuestions) {
    const id = readString(rawQuestion, "id");
    const question = readString(rawQuestion, "question");
    if (id === undefined || question === undefined) {
      return undefined;
    }
    const options: AskWireOption[] = [];
    for (const rawOption of readArray(rawQuestion, "options") ?? []) {
      const label = readString(rawOption, "label");
      if (label === undefined) {
        continue;
      }
      const description = readString(rawOption, "description");
      options.push(description === undefined ? { label } : { label, description });
    }
    const header = readString(rawQuestion, "header");
    questions.push({
      id,
      question,
      ...(header === undefined ? {} : { header }),
      multi_select: readBool(rawQuestion, "multi"),
      free_text: true,
      options,
    });
  }
  return questions;
}

/** 原生保留行标签（ask.ts:42-50）：命中说明这次调用只该由原生校验去拒绝。 */
const RESERVED_ASK_LABELS = new Set(["Other (type your own)", "Chat about this", "Next →"]);

/**
 * 这次调用是否该交给原生做校验。
 *
 * 原生 execute 内置三类 fail-closed 校验（保留标签冲突、问题 id 重复、同题选项重复，
 * ask.ts:844-921），影子路径不做这些校验就会把「居然被接受了的畸形调用」答出去，
 * 语义与终端不一致。判据命中即退回原生独占，由原生给出它自己那句错误文本——
 * 校验只有一处实现，不抄第二遍。
 */
function needsNativeAskValidation(questions: AskWireQuestion[]): boolean {
  const seenIds = new Set<string>();
  for (const question of questions) {
    if (seenIds.has(question.id)) {
      return true;
    }
    seenIds.add(question.id);
    const seenLabels = new Set<string>();
    for (const option of question.options) {
      if (RESERVED_ASK_LABELS.has(option.label) || seenLabels.has(option.label)) {
        return true;
      }
      seenLabels.add(option.label);
    }
  }
  return false;
}

/** 一题的作答：原生 `QuestionResult` 的形状（选中项与自由文本分开）。 */
interface AskAnswerItem {
  id: string;
  question: string;
  options: string[];
  multi: boolean;
  selectedOptions: string[];
  customInput?: string;
}

/**
 * 把刘海的作答折成原生形状：命中原生选项 label 的值进 `selectedOptions`，
 * 其余算自由文本（原生的 `customInput`，等价终端里的「Other」行）。
 */
function toAskAnswers(questions: AskWireQuestion[], answers: AskAnswers): AskAnswerItem[] {
  const items: AskAnswerItem[] = [];
  for (const question of questions) {
    const labels = question.options.map((option) => option.label);
    const chosen = (answers[question.id] ?? []).filter((value) => typeof value === "string" && value.length > 0);
    const selectedOptions = chosen.filter((value) => labels.includes(value));
    const customValues = chosen.filter((value) => !labels.includes(value));
    items.push({
      id: question.id,
      question: question.question,
      options: labels,
      multi: question.multi_select,
      selectedOptions,
      ...(customValues.length === 0 ? {} : { customInput: customValues.join("\n") }),
    });
  }
  return items;
}

/** 多题作答里的一行；文案对齐原生 `formatQuestionResult`（ask.ts:682-696）。 */
function formatAskAnswerLine(item: AskAnswerItem): string {
  if (item.customInput === undefined) {
    if (item.selectedOptions.length > 0) {
      return item.multi ? `${item.id}: [${item.selectedOptions.join(", ")}]` : `${item.id}: ${item.selectedOptions[0]}`;
    }
    return item.multi ? `${item.id}: []` : `${item.id}: (cancelled)`;
  }
  return `${item.id}: "${item.customInput}"`;
}

/** 单题作答的正文；文案对齐原生 `formatSingleQuestionResponse`（ask.ts:696-733）。 */
function formatSingleAskAnswer(item: AskAnswerItem): string {
  const parts: string[] = [];
  if (item.selectedOptions.length > 0) {
    parts.push(
      item.multi
        ? `User selected: ${item.selectedOptions.join(", ")}`
        : `User selected: ${item.selectedOptions[0]}`,
    );
  }
  if (item.customInput !== undefined) {
    const indented = item.customInput
      .split("\n")
      .map((line) => `  ${line}`)
      .join("\n");
    parts.push(
      item.customInput.includes("\n")
        ? `User provided custom input:\n${indented}`
        : `User provided custom input: ${item.customInput}`,
    );
  }
  if (parts.length > 0) {
    return parts.join("\n");
  }
  return item.multi ? "User did not select any options" : "User cancelled the selection";
}

/**
 * 组装刘海作答的工具结果。
 *
 * `content` 是模型看到的东西（与原生同一套文案），`details` 沿用原生
 * `AskToolDetails` 的形状（单题是扁平字段、多题是 `results`），TUI 渲染因此与
 * 终端作答一致。
 */
/**
 * 该题在这次作答里有没有出现（键存在，值可以是空数组 = 零选）。
 * 未出现的题沿用既有语义（按 items 里已作答的题组装）。
 */
function buildIslandAskResult(questions: AskWireQuestion[], answers: AskAnswers): AskToolResultLike {
  const items = toAskAnswers(questions, answers);
  if (items.length === 1) {
    const item = items[0] as AskAnswerItem;
    return {
      content: [{ type: "text", text: formatSingleAskAnswer(item) }],
      details: {
        question: item.question,
        options: item.options,
        multi: item.multi,
        selectedOptions: item.selectedOptions,
        ...(item.customInput === undefined ? {} : { customInput: item.customInput }),
      },
    };
  }
  return {
    content: [
      {
        type: "text",
        text: `User answers:\n${items.map(formatAskAnswerLine).join("\n")}`,
      },
    ],
    details: {
      results: items.map((item) => ({
        id: item.id,
        question: item.question,
        options: item.options,
        multi: item.multi,
        selectedOptions: item.selectedOptions,
        ...(item.customInput === undefined ? {} : { customInput: item.customInput }),
      })),
    },
  };
}

/** 抛一个与 omp `ToolAbortError` 同名同文案的错误（扩展不 import omp 内部模块）。 */
function throwAbortError(message: string): never {
  const error = new Error(message);
  error.name = "ToolAbortError";
  throw error;
}

/** 原生 headless 拒绝：`context.abort()` 后抛错，逐字对齐 ask.ts:832-834。 */
function refuseAskWithoutUI(context: ExtensionContextLike | undefined): never {
  try {
    (context as ShadowAskContextLike | undefined)?.abort?.();
  } catch {
    // 忽略：拿不到 abort 不影响「拒绝」这个结论。
  }
  throwAbortError("Ask tool requires interactive mode");
}

/** 用户放弃作答：撤下终端询问后，按原生取消语义终止本轮（ask.ts:1016 / 1125）。 */
function cancelAskFromIsland(context: ExtensionContextLike | undefined): never {
  try {
    (context as ShadowAskContextLike | undefined)?.abort?.();
  } catch {
    // 忽略：与原生一致——abort 拿不到也照抛，只是本轮不会提前结束。
  }
  throwAbortError("Ask tool was cancelled by the user");
}

/** 走原生一条路（终端的提问界面），结果原样返回。 */
async function takeNativeAskResult(
  native: NonNullable<ShadowAskContextLike["invokeTool"]>,
  params: Record<string, unknown>,
  options: { signal?: AbortSignal; onUpdate?: unknown },
): Promise<AskToolResultLike> {
  return await native(params, options);
}

/** 影子 ask 的入参形状（未解析的原始对象）。 */
type ShadowAskParams = Record<string, unknown>;

/**
 * 并发执行两条路并取先答者；返回值**就是**要交给模型的结果。
 *
 * - 刘海先答：abort 原生那条（撤下终端对话框），返回刘海作答；
 * - 刘海明确放弃（`deny`）：撤下终端对话框，按原生取消语义终止本轮；
 * - 原生先答：关掉刘海那条 socket（应用据此把卡片当外部已裁决），返回原生结果；
 * - 刘海超时 / 应用不可达 / 连接中断：不碰终端那条，等原生（功能不退化）。
 */
async function raceIslandAndNativeAsk(args: {
  toolCallId: string;
  params: ShadowAskParams;
  questions: AskWireQuestion[];
  native: NonNullable<ShadowAskContextLike["invokeTool"]>;
  signal?: AbortSignal;
  onUpdate?: unknown;
  context: ShadowAskContextLike;
}): Promise<AskToolResultLike> {
  const { toolCallId, params, questions, native, signal, onUpdate, context } = args;
  const body = buildBody(
    {
      event: "ToolApproval",
      status: "waiting_for_approval",
      expects_response: true,
      tool: "ask",
      tool_use_id: toolCallId,
      message: questions[0]?.question,
      ask: { questions },
    },
    { sessionId: readSessionIdFrom(context), sessionFile: currentSessionFilePath(context) },
  );
  // 拿不到会话身份就发不出可回答的卡片：退回原生独占（与应用侧「没有 tool_use_id 不登记待批」一致）。
  if (body === undefined) {
    return await takeNativeAskResult(native, params, { signal, onUpdate });
  }

  const island = startDecisionRequest(body, ASK_TIMEOUT_MS);
  // 本轮被中断（用户 Esc / 上层取消）时收掉 socket，别把 fd 吊在那儿。
  const onAbort = () => island.cancel();
  signal?.addEventListener("abort", onAbort);
  const nativeController = new AbortController();
  // 两条路都先折成「结局对象」：原生那条必须就地 catch，否则败者会变成未处理拒绝。
  const nativeRace = native(params, { signal: nativeController.signal, onUpdate })
    .then((result) => ({ from: "native" as const, result }))
    .catch((error: unknown) => ({ from: "native-error" as const, error }));
  const settleNative = async (): Promise<AskToolResultLike> => {
    const settled = await nativeRace;
    if (settled.from === "native-error") {
      throw settled.error;
    }
    return settled.result;
  };
  try {
    const first = await Promise.race([
      nativeRace,
      island.outcome.then((outcome) => ({ from: "island" as const, outcome })),
    ]);
    if (first.from !== "island") {
      // 原生先答：关掉刘海那条 socket，让应用把卡片当「外部已裁决」处理。
      island.cancel();
      return await settleNative();
    }
    // 判定「刘海是否真的作答」：**键存在即已作答**，值可以是空数组（多选零选）。
    // 这里若要求「非空」就会把零选误判成「没答」，落进下面的取消分支 →
    // `ctx.abort()` + ToolAbortError，把本该「正常完成」的零选变成终止整轮。
    const answers =
      first.outcome.kind === "decision" && first.outcome.decision === "answer"
        ? first.outcome.answers
        : undefined;
    if (answers !== undefined && Object.keys(answers).length > 0) {
      nativeController.abort();
      return buildIslandAskResult(questions, answers);
    }
    if (first.outcome.kind === "decision") {
      nativeController.abort();
      cancelAskFromIsland(context);
    }
    // 超时 / 应用不可达 / 连接中断：用户还可能就在终端前，等原生那条给答案。
    return await settleNative();
  } finally {
    signal?.removeEventListener("abort", onAbort);
  }
}

/** 影子 `ask` 的 zod 参数 schema；与原生 arkType schema 同形（ask.ts:51-74）。 */
function buildAskShadowSchema(zod: ExtensionZodLike): unknown {
  const option = zod.object({
    label: zod.string().describe("display label"),
    description: zod.string().optional().describe("optional explanatory text displayed below the label"),
    preview: zod.string().optional().describe("optional rich preview content for interactive ask dialogs"),
  });
  const question = zod.object({
    id: zod.string().describe("question id"),
    question: zod.string().describe("question text"),
    header: zod.string().optional().describe("optional short display chip for rich ask dialogs"),
    options: zod.array(option).describe("available options"),
    multi: zod.boolean().optional().describe("allow multiple selections"),
    recommended: zod.number().optional().describe("recommended option index"),
  });
  return zod.object({ questions: zod.array(question).min(1).describe("questions to ask") });
}

/** 影子 `ask` 给模型看的描述：与原生 prompts/tools/ask.md 同一套规则（浓缩）。 */
const SHADOW_ASK_DESCRIPTION = [
  "Ask user for clarification/input during task execution.",
  "",
  "<conditions>",
  "- Multiple approaches with significantly different tradeoffs user should weigh.",
  "</conditions>",
  "",
  "<instruction>",
  "- `recommended: <index>` marks default (0-indexed); \" (Recommended)\" added automatically.",
  "- Use `questions` for related questions, not one at a time.",
  "- Set `multi: true` on a question to allow multiple selections.",
  "- Short option labels; explanatory tradeoffs in `description`, not labels.",
  "</instruction>",
  "",
  "<caution>",
  "- Provide 2-5 concise, distinct options.",
  "</caution>",
  "",
  "<critical>",
  "- Default to action. Resolve ambiguity via repo conventions, existing patterns, reasonable defaults.",
  "- If multiple choices acceptable: pick most conservative/standard option; proceed; state choice.",
  "- Do NOT include \"Other\"; the UI adds \"Other (type your own)\" to every question automatically.",
  "</critical>",
].join("\n");

/**
 * 注册影子 `ask`（只在 omp 且用户没关掉 ask 时）。
 *
 * `approval: "read"` 与原生一致（ask.ts:748），`strict` 与 `label`/`name` 也照抄原生，同时本文件的 `NEVER_ASK_TOOLS` 也把
 * `ask` 列为自动放行——两处一致，否则自己的 `tool_call` 闸门会把这次提问再拦一次。
 * `concurrency: "exclusive"` 与原生一致（ask.ts:799）：交互提问独占用一批，
 * 否则两条 ask 会同时抢刘海卡片与终端对话框。该字段不在 `ToolDefinition` 的公开
 * 形状里，但 omp 会把它从定义对象上按原型链透传到注册表项（applyToolProxy，
 * tool-proxy.ts:5-35；调度处 agent-loop.ts:3338-3356）。
 */
function registerShadowAsk(pi: ExtensionApi): void {
  if (AGENT !== "omp") {
    // pi 0.85.1 没有 `ask` 工具（dist/core/tools 只有 bash/edit/find/grep/ls/read/write/powershell），
    // 注册同名影子只会给模型凭空多出一个永远失败的 ask。
    return;
  }
  const register = pi.registerTool;
  const zod = pi.zod;
  if (typeof register !== "function" || !zod) {
    return;
  }
  if (readAskEnabled(pi) === false) {
    // 用户显式关了 ask（settings `ask.enabled`，默认 true）：不把它加回来。
    return;
  }
  try {
    register.call(pi, {
      name: "ask",
      label: "Ask",
      description: SHADOW_ASK_DESCRIPTION,
      parameters: buildAskShadowSchema(zod),
      approval: "read",
      strict: true,
      loadMode: "discoverable",
      concurrency: "exclusive",
      async execute(
        toolCallId: string,
        params: ShadowAskParams,
        signal?: AbortSignal,
        onUpdate?: unknown,
        ctx?: ShadowAskContextLike,
      ): Promise<AskToolResultLike> {
        const native = ctx?.invokeTool;
        const questions = normalizeAskQuestions(params);
        // 没有原生可委托：headless、子会话、老版 omp 没有 ask、或用户关了 ask.enabled。
        // 按原生语义拒绝（abort + ToolAbortError），绝不假装问过。
        if (typeof native !== "function") {
          refuseAskWithoutUI(ctx);
        }
        // 只有「TUI 根会话」才走刘海：headless 与子 Agent 交给原生（原生自己会拒绝），
        // 免得在没有人看的会话里发一张点不动的卡片。
        if (
          questions === undefined ||
          needsNativeAskValidation(questions) ||
          ctx?.hasUI !== true ||
          !isRootSession(ctx)
        ) {
          return await takeNativeAskResult(native, params, { signal, onUpdate });
        }
        return await raceIslandAndNativeAsk({
          toolCallId,
          params,
          questions,
          native,
          signal,
          onUpdate,
          context: ctx,
        });
      },
    });
    shadowAskRegistered = true;
  } catch {
    // 注册失败（老版 omp 没有 registerTool）：保持只上报/闸门行为，绝不因此抛错。
  }
}

/** 读 settings 里的 `ask.enabled`；读不到返回 undefined（按原生默认 true 处理）。 */
function readAskEnabled(pi: ExtensionApi): boolean | undefined {
  try {
    const settings = (pi as { pi?: { settings?: { get?(key: string): unknown } } }).pi?.settings;
    if (!settings || typeof settings.get !== "function") {
      return undefined;
    }
    const value = settings.get("ask.enabled");
    return typeof value === "boolean" ? value : undefined;
  } catch {
    return undefined;
  }
}

export default function (pi: ExtensionApi) {
  // 启动自检：扩展 API 形状不对就降级为只上报，并让应用知道「闸门没在用」。
  const probeFailure = probeGateSupport(pi);
  if (probeFailure) {
    setGateEnabled(false, probeFailure);
  }

  // 影子 `ask`：把 omp 的交互式提问接到刘海（只在 omp 且用户没关 ask 时注册）。
  registerShadowAsk(pi);

  /** 只有根会话才绑定身份并上报会话事件；子代理上下文只读不写（见模块级变量注释）。 */
  function activateRootSession(ctx: ExtensionContextLike | undefined): boolean {
    if (!isRootSession(ctx)) {
      return false;
    }
    bindRootSession(ctx);
    return true;
  }

  pi.on("session_start", (_event, ctx) => {
    if (!activateRootSession(ctx)) {
      return;
    }
    if (pid === undefined) {
      pid = process.pid;
      tty = detectTty();
    }
    // 只读诊断：omp 生效的审批档位（用于应用侧做「双提示」告警）。
    approvalMode = readApprovalMode(pi);
    // 扩展可能在会话中途被加载：按上下文判断是否正在跑。
    const busy = ctx?.isIdle?.() === false;
    void send({
      event: "SessionStart",
      status: busy ? "processing" : "idle",
      gate_enabled: gateEnabled,
      gate_disabled_reason: gateDisabledReason,
      integration_version: EXTENSION_VERSION,
      degradation: DEGRADATION,
      approval_mode: approvalMode,
    });
    if (!gateEnabled) {
      reportGateDisabledOnce();
    }
  });

  pi.on("session_switch", (_event, ctx) => {
    if (!isRootSession(ctx)) {
      return;
    }
    // 同一个进程切走的旧会话不会再被写盘，而进程还活着、存活回收不会触发；
    // 若只发 SessionStart，旧会话行会一直挂在列表里，因此先给旧会话补一个结束事件。
    const previousSessionId = rootSessionId;
    const previousSessionFile = rootSessionFile;
    bindRootSession(ctx);
    if (previousSessionId && previousSessionId !== rootSessionId) {
      void send(
        { event: "SessionEnd", status: "ended" },
        { sessionId: previousSessionId, sessionFile: previousSessionFile },
      );
    }
    void send({ event: "SessionStart", status: "idle" });
  });

  pi.on("agent_start", (_event, ctx) => {
    if (!activateRootSession(ctx)) {
      return;
    }
    void send({ event: "UserPromptSubmit", status: "processing" });
  });

  pi.on("agent_end", (event, ctx) => {
    if (ctx !== undefined && !isRootSession(ctx)) {
      return;
    }
    // willContinue 表示还会继续（自动重试/续跑），此时不算一轮结束。
    if (readBool(event, "willContinue")) {
      return;
    }
    void send({ event: "Stop", status: "waiting_for_input" });
  });

  // Pi 侧的生命周期事件名（与 omp 的 agent_end 等价）；不支持时忽略。
  try {
    pi.on("agent_settled", (_event, ctx) => {
      if (ctx !== undefined && !isRootSession(ctx)) {
        return;
      }
      if (ctx?.isIdle?.() !== true) {
        return;
      }
      void send({ event: "Stop", status: "waiting_for_input" });
    });
  } catch {
    // 该事件在部分版本不存在。
  }

  // 审批闸门（核心）：先按档位定级，需要问的阻塞等刘海决定。
  pi.on("tool_call", async (event, ctx) => {
    const toolName = readString(event, "toolName") ?? "";
    const toolCallId = readString(event, "toolCallId");
    const input =
      typeof event === "object" && event !== null
        ? (event as Record<string, unknown>).input
        : undefined;
    let tier: ApprovalTier = "exec";
    try {
      // 普通上报保持原样：走 chain、fire-and-forget（卡片要尽快出现）。
      if (isRootSession(ctx)) {
        void send({
          event: "PreToolUse",
          status: "running_tool",
          tool: toolName,
          tool_input: input,
          tool_use_id: toolCallId,
        });
      }

      tier = classifyToolCall(toolName, input);
      if (tier === "allow") {
        return;
      }
      // 「只问危险命令」档：写档与执行档照跑——上面那条 PreToolUse 已经上报过，
      // 行里仍看得到工具在跑，只是不再等刘海点按。
      if (!asksByScope(tier)) {
        return;
      }
      // 子会话默认不拦（`task` 本身已是授权边界，逐条上卡会刷屏并阻塞子代理）；危险命令例外。
      if (!isGateRootSession(ctx) && tier !== "critical") {
        return;
      }
      if (!toolCallId) {
        // 没有工具调用 id 就无法让应用登记待批：降级为只上报，绝不「看起来问了」。
        setGateEnabled(false, "tool_call without toolCallId");
        reportGateDisabledOnce();
        return;
      }
      const body = buildBody(
        {
          event: "ToolApproval",
          status: "waiting_for_approval",
          expects_response: true,
          tool: toolName,
          tool_input: input,
          tool_use_id: toolCallId,
          approval_kind: tier,
          parent_session: hasParentSession(ctx),
          gate_enabled: gateEnabled,
          degradation: DEGRADATION,
          approval_mode: approvalMode,
        },
        { sessionId: readSessionIdFrom(ctx), sessionFile: currentSessionFilePath(ctx) },
      );
      if (!body) {
        // 拿不到会话身份：这一条只上报，不做闸门化（与既有 fail-open 一致）。
        return;
      }
      const outcome = await requestDecision(body, APPROVAL_TIMEOUT_MS);
      const verdict = resolveOutcome(tier, outcome);
      if (verdict.allowed) {
        if (outcome.kind === "unreachable") {
          // 降级放行：TUI 留一行可见提示 + 排队一条审计事件（应用回来后补投）。
          const notice = `AgentIsland gate offline — degraded to ${DEGRADATION}`;
          if (ctx?.hasUI === true) {
            notifyGateDegraded(ctx, notice);
          }
          recordDegradedReport({
            ...body,
            status: "degraded_allowed",
            expects_response: false,
            gate_notice: notice,
          });
        }
        return;
      }
      const blocked: ToolCallVerdict = {
        block: true,
        reason: verdict.reason ?? "Denied on AgentIsland",
      };
      return blocked;
    } catch {
      // 闸门永不抛错：抛错会被 omp 记成扩展失败并阻塞工具。危险命令仍然拒绝。
      if (tier === "critical") {
        return { block: true, reason: "Blocked on AgentIsland: gate error on a known-dangerous command" };
      }
      return;
    }
  });

  pi.on("tool_result", (event, ctx) => {
    if (!isRootSession(ctx)) {
      return;
    }
    const failed = readBool(event, "isError");
    void send({
      event: failed ? "PostToolUseFailure" : "PostToolUse",
      status: "processing",
      tool: readString(event, "toolName"),
      tool_use_id: readString(event, "toolCallId"),
    });
  });

  // OMP 自己的审批门在响（非 yolo 的 exec 档、显式 prompt 策略、computer safetyChecks）。
  // 上报让位信号：应用据此撤下自己的 Allow/Deny，只显示「终端正在询问」。
  pi.on("tool_approval_requested", (event, ctx) => {
    if (!isRootSession(ctx)) {
      return;
    }
    void send({
      event: "ToolApproval",
      status: "waiting_for_approval",
      expects_response: false,
      omp_owns_approval: true,
      tool: readString(event, "toolName"),
      tool_use_id: readString(event, "toolCallId"),
      message: readString(event, "reason"),
    });
  });

  pi.on("tool_approval_resolved", (event, ctx) => {
    if (!isRootSession(ctx)) {
      return;
    }
    const approved = readBool(event, "approved");
    void send({
      // 被拒绝的工具不会产生 tool_result（omp 的审批门在 execute 之前），
      // 沿用 PostToolUse 会把「拒绝」显示成成功，因此按结果分流。
      event: approved ? "PostToolUse" : "PostToolUseFailure",
      status: "processing",
      tool: readString(event, "toolName"),
      tool_use_id: readString(event, "toolCallId"),
    });
  });

  // `ask` 工具会阻塞等待用户回答，同样属于「等待输入」。
  pi.on("tool_execution_start", (event, ctx) => {
    if (readString(event, "toolName") !== "ask" || !isRootSession(ctx)) {
      return;
    }
    // 影子工具在场时，可作答的卡片由它自己在**真的去问刘海**那一刻发出（带
    // expects_response 与完整 questions）；这里再发一条会在应用侧留下「只展示、
    // 点不动」的重复卡，并把可作答的那张挤掉。
    if (shadowAskRegistered) {
      return;
    }
    const args = typeof event === "object" && event !== null ? (event as Record<string, unknown>).args : undefined;
    const questions =
      typeof args === "object" && args !== null && Array.isArray((args as Record<string, unknown>).questions)
        ? ((args as Record<string, unknown>).questions as unknown[])
        : [];
    const first = questions.find((q) => typeof readString(q, "question") === "string");
    void send({
      event: "ToolApproval",
      status: "waiting_for_approval",
      tool: "ask",
      tool_use_id: readString(event, "toolCallId"),
      message: readString(first, "question"),
    });
  });

  pi.on("tool_execution_end", (event, ctx) => {
    if (readString(event, "toolName") !== "ask" || !isRootSession(ctx)) {
      return;
    }
    void send({
      event: "PostToolUse",
      status: "processing",
      tool: "ask",
      tool_use_id: readString(event, "toolCallId"),
    });
  });

  pi.on("session_shutdown", () => {
    void send({ event: "SessionEnd", status: "ended" });
  });

  // 子 Agent：由父会话的共享总线代为上报（见 subscribeSubagentBus 注释）。
  subscribeSubagentBus(pi);
}
